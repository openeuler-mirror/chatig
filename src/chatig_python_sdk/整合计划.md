### 2. **模块注册机制**
```python
# 在统一SDK中添加模块
def _initialize_modules(self):
    """初始化各个模块"""
    self.modules = {}
    
    # Chat模块
    if self.config.get('chat_module', {}).get('enabled', False):
        try:
            from chat_sdk.chat_client import ChatClient
            from chat_sdk.config import ChatConfig
            
            chat_config = ChatConfig()
            # 应用配置...
            self.modules['chat'] = ChatClient(config=chat_config)
        except Exception as e:
            print(f"Chat模块初始化失败: {e}")
    
    # Embeddings模块
    if self.config.get('embeddings_module', {}).get('enabled', False):
        try:
            from embeddings_sdk.embeddings_client import EmbeddingsClient
            from embeddings_sdk.config import EmbeddingsConfig
            
            embeddings_config = EmbeddingsConfig()
            # 应用配置...
            self.modules['embeddings'] = EmbeddingsClient(config=embeddings_config)
        except Exception as e:
            print(f"Embeddings模块初始化失败: {e}")
    
    # ... 其他模块
```
